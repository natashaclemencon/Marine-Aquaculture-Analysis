# Create function 

suitable <- function(minsst, maxsst, mindepth, maxdepth, species_name)
{
# Create a matrix for the growth levels
rcldepth <- matrix(c(
   -Inf, maxdepth, 0, 
   maxdepth,   mindepth,  1,
   mindepth,  Inf,  0
), ncol = 3, byrow = TRUE)

# Matrix for temperature
rcltemp <- matrix(c(
   -Inf,   minsst,  0,
   minsst,  maxsst,  1,
  maxsst,  Inf,  0
), ncol = 3, byrow = TRUE)

# Create rasters for suitable temp and suitable depth.
suitabledepth <- (classify(croppeddepth, rcl = rcldepth))
suitabletemp <- (classify(fiveyearmeancelsius, rcl = rcltemp))

# Create a vector of suitable depth and suitable temp. 
suitablezones <- c(suitabledepth, suitabletemp)

# Find suitable areas for a species with lapp()
suitableareas <- lapp(suitablezones, fun = function(s, d) s*d)

# Make suitableareas a raster
eezraster <- rasterize(eezboundaries, suitableareas, field="rgn_id")

# Use cellSize function to find the total area within each EEZ zone.
suitableareascellsize <- cellSize(suitableareas, unit="km", mask=TRUE)

# Mask eez to remove NAs and unsuitable zones. 
eezmasked <- mask(suitableareascellsize, suitableareas, maskvalue=0)

# Find the suitable zones in each region.
zone <- zonal(eezmasked, eezraster, fun = "sum", na.rm=TRUE)

# To make the map, first join the suitable regions by zone with the economic zone boundaries.
map <- left_join(eezboundaries, zone, by="rgn_id")

# Plot it!
plot <- tm_shape(map)+
tm_basemap("OpenStreetMap") +
    tm_polygons() +
    tm_fill(
        fill='area'
    )+
tm_title(paste("Suitable", species_name, "Habitats"))
tm_text(text='rgn', size = 0.35)
tm_add_legend(type="polygons"
)

return(plot)
}
